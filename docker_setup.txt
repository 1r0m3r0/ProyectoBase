version: '3.8'

services:
  # Bot de Trading
  trading-bot:
    build: .
    container_name: trading_bot
    restart: unless-stopped
    env_file:
      - .env
    environment:
      - MODE=${MODE:-testnet}
      - DATABASE_TYPE=${DATABASE_TYPE:-csv}
    volumes:
      - ./data:/app/data
      - ./logs:/app/logs
      - ./config:/app/config
    networks:
      - trading-network
    depends_on:
      - postgres
      - mongodb
    command: python main.py --mode ${MODE:-testnet}

  # Dashboard Streamlit
  dashboard:
    build: .
    container_name: trading_dashboard
    restart: unless-stopped
    env_file:
      - .env
    ports:
      - "8501:8501"
    volumes:
      - ./data:/app/data
      - ./logs:/app/logs
    networks:
      - trading-network
    command: streamlit run visualization/dashboard.py

  # API FastAPI para móvil
  api:
    build: .
    container_name: trading_api
    restart: unless-stopped
    env_file:
      - .env
    ports:
      - "8000:8000"
    volumes:
      - ./data:/app/data
    networks:
      - trading-network
    command: uvicorn visualization.mobile_api.fastapi_server:app --host 0.0.0.0 --port 8000

  # PostgreSQL (opcional)
  postgres:
    image: postgres:15-alpine
    container_name: trading_postgres
    restart: unless-stopped
    environment:
      POSTGRES_DB: trading_bot
      POSTGRES_USER: ${POSTGRES_USER:-tradingbot}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-securepassword}
    volumes:
      - postgres_data:/var/lib/postgresql/data
    ports:
      - "5432:5432"
    networks:
      - trading-network

  # MongoDB (opcional)
  mongodb:
    image: mongo:7
    container_name: trading_mongodb
    restart: unless-stopped
    environment:
      MONGO_INITDB_ROOT_USERNAME: ${MONGO_USER:-tradingbot}
      MONGO_INITDB_ROOT_PASSWORD: ${MONGO_PASSWORD:-securepassword}
      MONGO_INITDB_DATABASE: trading_bot
    volumes:
      - mongodb_data:/data/db
    ports:
      - "27017:27017"
    networks:
      - trading-network

networks:
  trading-network:
    driver: bridge

volumes:
  postgres_data:
  mongodb_data:
